/** Prefijos que Excel/LibreOffice interpretan como fórmula (CSV injection). */
const FORMULA_PREFIX = /^[=+\-@\t\r]/;
const NEEDS_QUOTES = /[",\n\r]/;

/** Serializa un valor como celda CSV segura. Los números no se tocan (G/B pueden ser negativos). */
export function toCsvCell(value: unknown): string {
  if (value === null || value === undefined) return '';
  if (typeof value === 'number') return Number.isFinite(value) ? String(value) : '';
  if (typeof value === 'boolean') return value ? 'true' : 'false';
  if (value instanceof Date) return value.toISOString();

  let text = String(value);
  if (FORMULA_PREFIX.test(text)) text = `'${text}`;
  if (NEEDS_QUOTES.test(text)) text = `"${text.replace(/"/g, '""')}"`;
  return text;
}

export function toCsvRow(values: unknown[]): string {
  return values.map(toCsvCell).join(',');
}
```

import { toCsvRow } from './csv.util';

export const CSV_HEADER = [
  'measurement_id', 'taken_at', 'taken_by', 'preset', 'watcher_status',
  'frequency_hz', 'channel', 'channel_alias', 'g_s', 'b_s',
  'concrete_mix', 'has_emulsifier', 'sample_age', 'temperature', 'humidity',
];

export interface CsvChannel {
  channelNumber: number;
  channelAlias: string | null;
  concreteMix: number | null;
  hasEmulsifier: boolean | null;
  sampleAge: number | null;
  temperature: number | null;
  humidity: number | null;
}

export interface CsvMeasurement {
  id: number;
  takenAt: Date | null;
  takenByName: string | null;
  presetName: string | null;
  watcherStatus: string | null;
  data: unknown;
  channels: CsvChannel[];
}

interface StoredPoint {
  frequency: number;
  g?: number[];
  b?: number[];
}

/** Formato largo: una fila por (frecuencia, canal medido). Solo canales con fila en measurement_channels. */
export function measurementToRows(m: CsvMeasurement): unknown[][] {
  const points = Array.isArray(m.data) ? (m.data as StoredPoint[]) : [];
  const channels = [...m.channels].sort((a, b) => a.channelNumber - b.channelNumber);
  const rows: unknown[][] = [];

  for (const point of points) {
    for (const ch of channels) {
      const idx = ch.channelNumber - 1;
      rows.push([
        m.id,
        m.takenAt ? m.takenAt.toISOString() : '',
        m.takenByName ?? '',
        m.presetName ?? '',
        m.watcherStatus ?? '',
        point.frequency,
        ch.channelNumber,
        ch.channelAlias ?? '',
        point.g?.[idx] ?? null,
        point.b?.[idx] ?? null,
        ch.concreteMix,
        ch.hasEmulsifier,
        ch.sampleAge,
        ch.temperature,
        ch.humidity,
      ]);
    }
  }
  return rows;
}

/** CSV completo: BOM (para que Excel lea tildes), metadatos `# ...`, encabezado y filas. */
export function buildCsv(metaLines: string[], measurements: CsvMeasurement[]): string {
  const lines = [
    ...metaLines.map((l) => `# ${l.replace(/[\r\n]+/g, ' ')}`),
    CSV_HEADER.join(','),
    ...measurements.flatMap((m) => measurementToRows(m).map(toCsvRow)),
  ];
  return '\uFEFF' + lines.join('\r\n');
}
```

import { BadRequestException, Injectable, NotFoundException } from '@nestjs/common';
import { Prisma } from '@prisma/client';
import { PrismaService } from '../prisma/prisma.service';
import { TeamScopeService } from '../common/team-scope.service';
import { buildCsv, CsvChannel, CsvMeasurement } from './measurement-csv';

export interface DateRange { from: Date | null; to: Date | null }

const channelSelect = {
  channelNumber: true, channelAlias: true, concreteMix: true, hasEmulsifier: true,
  sampleAge: true, temperature: true, humidity: true,
} satisfies Prisma.MeasurementChannelSelect;

type RawChannel = Prisma.MeasurementChannelGetPayload<{ select: typeof channelSelect }>;

const num = (d: Prisma.Decimal | null): number | null => (d === null ? null : Number(d));

function toCsvChannel(c: RawChannel): CsvChannel {
  return {
    channelNumber: c.channelNumber,
    channelAlias: c.channelAlias,
    concreteMix: num(c.concreteMix),
    hasEmulsifier: c.hasEmulsifier,
    sampleAge: c.sampleAge,
    temperature: num(c.temperature),
    humidity: num(c.humidity),
  };
}

function slug(text: string): string {
  return text.normalize('NFD').replace(/[\u0300-\u036f]/g, '')
    .replace(/[^a-zA-Z0-9]+/g, '-').replace(/^-|-$/g, '').toLowerCase() || 'tarea';
}

function yyyymmdd(d = new Date()): string {
  return d.toISOString().slice(0, 10).replace(/-/g, '');
}

@Injectable()
export class ReportsService {
  constructor(private prisma: PrismaService, private teamScope: TeamScopeService) {}

  /** Valida y convierte `from`/`to` (ISO). `to` sin hora incluye el día completo. */
  parseRange(from?: string, to?: string): DateRange {
    const parse = (value: string | undefined, endOfDay: boolean): Date | null => {
      if (!value) return null;
      const date = new Date(value);
      if (Number.isNaN(date.getTime())) throw new BadRequestException(`Fecha inválida: ${value}`);
      if (endOfDay && /^\d{4}-\d{2}-\d{2}$/.test(value)) date.setUTCHours(23, 59, 59, 999);
      return date;
    };
    const range = { from: parse(from, false), to: parse(to, true) };
    if (range.from && range.to && range.from > range.to) {
      throw new BadRequestException('La fecha "desde" debe ser anterior a "hasta"');
    }
    return range;
  }

  private takenAtWhere(range: DateRange): Prisma.MeasurementWhereInput {
    if (!range.from && !range.to) return {};
    return { takenAt: { ...(range.from ? { gte: range.from } : {}), ...(range.to ? { lte: range.to } : {}) } };
  }

  async listTasks(userId: string, projectId?: string) {
    const teamWhere = await this.teamScope.projectWhereForUser(userId);
    if (projectId) await this.teamScope.assertProjectInTeam(userId, projectId);

    const tasks = await this.prisma.task.findMany({
      where: { project: { ...teamWhere, ...(projectId ? { uuid: projectId } : {}) } },
      select: {
        id: true, name: true, status: true,
        project: { select: { uuid: true, name: true } },
        measurements: { select: { takenAt: true, watcherStatus: true } },
      },
    });

    return tasks
      .map((t) => {
        const statusCounts = { correct: 0, warning: 0, error: 0 };
        let last: Date | null = null;
        for (const m of t.measurements) {
          if (m.watcherStatus === 'correct' || m.watcherStatus === 'warning' || m.watcherStatus === 'error') {
            statusCounts[m.watcherStatus]++;
          }
          if (m.takenAt && (!last || m.takenAt > last)) last = m.takenAt;
        }
        return {
          taskId: t.id,
          taskName: t.name,
          taskStatus: t.status,
          projectId: t.project.uuid,
          projectName: t.project.name,
          measurementCount: t.measurements.length,
          lastMeasuredAt: last ? last.toISOString() : null,
          statusCounts,
        };
      })
      .sort((a, b) => (b.lastMeasuredAt ?? '').localeCompare(a.lastMeasuredAt ?? ''));
  }

  async getTaskReport(userId: string, taskId: number, range: DateRange) {
    await this.teamScope.assertTaskInTeam(userId, taskId);

    const task = await this.prisma.task.findUnique({
      where: { id: taskId },
      select: {
        id: true, name: true, description: true,
        project: { select: { uuid: true, name: true } },
        taskPreferences: {
          take: 1,
          select: {
            name: true,
            analyzerDevice: { select: { name: true, ip: true } },
            muxDevice: { select: { name: true, ip: true } },
            ranges: { orderBy: { order: 'asc' }, select: { startFreq: true, stopFreq: true, points: true, sweepType: true } },
          },
        },
      },
    });
    if (!task) throw new NotFoundException('Task not found');

    const measurements = await this.prisma.measurement.findMany({
      where: { taskId, ...this.takenAtWhere(range) },
      orderBy: { takenAt: 'asc' },
      select: {
        id: true, takenAt: true, takenByName: true, presetName: true, sweepType: true,
        watcherStatus: true, watcherReason: true,
        measurementChannels: { select: channelSelect },
      },
    });

    const lastId = measurements.length ? measurements[measurements.length - 1].id : null;
    const latest = lastId
      ? await this.prisma.measurement.findUnique({
          where: { id: lastId },
          select: { id: true, takenAt: true, sweepType: true, data: true, measurementChannels: { select: channelSelect } },
        })
      : null;

    const pref = task.taskPreferences[0];
    return {
      task: { id: task.id, name: task.name, description: task.description },
      project: { id: task.project.uuid, name: task.project.name },
      preference: pref
        ? {
            name: pref.name,
            analyzer: pref.analyzerDevice,
            mux: pref.muxDevice,
            ranges: pref.ranges.map((r) => ({
              startFreq: Number(r.startFreq), stopFreq: Number(r.stopFreq), points: r.points, sweepType: r.sweepType,
            })),
          }
        : null,
      measurements: measurements.map(({ measurementChannels, ...m }) => ({
        ...m,
        channels: measurementChannels.map(toCsvChannel),
      })),
      latest: latest
        ? {
            id: latest.id,
            takenAt: latest.takenAt,
            sweepType: latest.sweepType,
            data: Array.isArray(latest.data) ? latest.data : [],
            channels: latest.measurementChannels.map(toCsvChannel),
          }
        : null,
      range: { from: range.from?.toISOString() ?? null, to: range.to?.toISOString() ?? null },
    };
  }

  private async loadCsvMeasurements(where: Prisma.MeasurementWhereInput): Promise<CsvMeasurement[]> {
    const rows = await this.prisma.measurement.findMany({
      where,
      orderBy: { takenAt: 'asc' },
      select: {
        id: true, takenAt: true, takenByName: true, presetName: true, watcherStatus: true, data: true,
        measurementChannels: { select: channelSelect },
      },
    });
    return rows.map(({ measurementChannels, ...m }) => ({ ...m, channels: measurementChannels.map(toCsvChannel) }));
  }

  private async taskHeader(taskId: number) {
    const task = await this.prisma.task.findUnique({
      where: { id: taskId },
      select: { name: true, project: { select: { name: true } } },
    });
    if (!task) throw new NotFoundException('Task not found');
    return task;
  }

  async measurementCsv(userId: string, measurementId: number): Promise<{ filename: string; content: string }> {
    await this.teamScope.assertMeasurementInTeam(userId, measurementId);
    const [measurement] = await this.loadCsvMeasurements({ id: measurementId });
    if (!measurement) throw new NotFoundException('Measurement not found');

    const { taskId } = await this.prisma.measurement.findUniqueOrThrow({ where: { id: measurementId }, select: { taskId: true } });
    const task = await this.taskHeader(taskId);

      [`Kronos | Proyecto: ${task.project.name} | Tarea: ${task.name}`, `Medición ${measurementId} | Generado: ${new Date().toISOString()}`],
      [measurement],
    );
    return { filename: `kronos_${slug(task.name)}_${measurementId}_${yyyymmdd()}.csv`, content };
  }

  async taskCsv(userId: string, taskId: number, range: DateRange): Promise<{ filename: string; content: string }> {

    );

  }

import apiClient from '../api/client';

export function saveBlob(blob: Blob, filename: string): void {
  const url = URL.createObjectURL(blob);
  const a = document.createElement('a');
  a.href = url;
  a.download = filename;
  document.body.appendChild(a);
  a.click();
  a.remove();
  setTimeout(() => URL.revokeObjectURL(url), 1000);
}

function filenameFrom(disposition: string | undefined, fallback: string): string {
  const match = disposition?.match(/filename="?([^";]+)"?/i);
  return match?.[1] ?? fallback;
}

/** GET autenticado (token del interceptor) que descarga la respuesta como archivo. */
export async function downloadFromApi(path: string, params?: Record<string, string | undefined>): Promise<string> {
  const res = await apiClient.get(path, { params, responseType: 'blob' });
  const filename = filenameFrom(res.headers['content-disposition'], 'kronos.csv');
  saveBlob(res.data as Blob, filename);
  return filename;
}

/** Con responseType 'blob', el cuerpo de error también llega como Blob: lo convierte a objeto. */
export async function readBlobError(error: unknown): Promise<unknown> {
  const data = (error as { response?: { data?: unknown } })?.response?.data;
  if (data instanceof Blob) {
    try { return JSON.parse(await data.text()); } catch { return undefined; }
  }
  return data;
}

export function consumePostLoginRedirect(fallback = '/dashboard'): string {
  try {
    const path = sessionStorage.getItem(KEY);
    sessionStorage.removeItem(KEY);
    return path && path.startsWith('/') && !path.startsWith('//') ? path : fallback;

export const formatFreq = (hz: number): string => {
  if (hz >= 1e6) return `${+(hz / 1e6).toFixed(3)} MHz`;
  if (hz >= 1e3) return `${+(hz / 1e3).toFixed(3)} kHz`;
  return `${+hz.toFixed(2)} Hz`;
};

```tsx
import { Svg, Path, Line, Text as SvgText, G } from '@react-pdf/renderer';
import type { ReportPoint } from '../../types/report.types';
import { channelColor, formatFreq, formatSci } from './format';

interface Props { data: ReportPoint[]; channels: number[]; log: boolean; width?: number; height?: number }

const PAD = { top: 10, right: 12, bottom: 28, left: 48 };

const MeasurementChartPdf = ({ data, channels, log, width = 515, height = 230 }: Props) => {
  const freqs = data.map((p) => p.frequency).filter((f) => f > 0);
  if (freqs.length < 2 || channels.length === 0) return null;

  const values = data.flatMap((p) => channels.flatMap((c) => [p.g[c - 1] ?? 0, p.b[c - 1] ?? 0]));
  let yMin = Math.min(...values);
  let yMax = Math.max(...values);
  if (yMin === yMax) { yMin -= 1; yMax += 1; }

  const fx = log ? Math.log10 : (v: number) => v;
  const xMin = fx(Math.min(...freqs));
  const xMax = fx(Math.max(...freqs));
  const plotW = width - PAD.left - PAD.right;
  const plotH = height - PAD.top - PAD.bottom;
  const sx = (f: number) => PAD.left + ((fx(f) - xMin) / (xMax - xMin || 1)) * plotW;
  const sy = (v: number) => PAD.top + (1 - (v - yMin) / (yMax - yMin)) * plotH;

  const path = (key: 'g' | 'b', ch: number) =>
    data
      .filter((p) => p.frequency > 0)
      .map((p, i) => `${i === 0 ? 'M' : 'L'}${sx(p.frequency).toFixed(1)},${sy(p[key][ch - 1] ?? 0).toFixed(1)}`)
      .join(' ');

  const yTicks = Array.from({ length: 5 }, (_, i) => yMin + ((yMax - yMin) * i) / 4);
  const xTicks = Array.from({ length: 5 }, (_, i) => {
    const t = xMin + ((xMax - xMin) * i) / 4;
    return log ? 10 ** t : t;
  });

  return (
    <Svg width={width} height={height}>
      {yTicks.map((v) => (
        <G key={`y${v}`}>
          <Line x1={PAD.left} x2={width - PAD.right} y1={sy(v)} y2={sy(v)} stroke="#e5e7eb" strokeWidth={0.5} />
          <SvgText x={PAD.left - 4} y={sy(v) + 2.5} style={{ fontSize: 6 }} fill="#374151" textAnchor="end">{formatSci(v)}</SvgText>
        </G>
      ))}
      {xTicks.map((f) => (
        <SvgText key={`x${f}`} x={sx(f)} y={height - PAD.bottom + 10} style={{ fontSize: 6 }} fill="#374151" textAnchor="middle">
          {formatFreq(f)}
        </SvgText>
      ))}
      <Line x1={PAD.left} x2={width - PAD.right} y1={PAD.top + plotH} y2={PAD.top + plotH} stroke="#9ca3af" strokeWidth={0.7} />
      <Line x1={PAD.left} x2={PAD.left} y1={PAD.top} y2={PAD.top + plotH} stroke="#9ca3af" strokeWidth={0.7} />
      <SvgText x={PAD.left + plotW / 2} y={height - 4} style={{ fontSize: 7 }} fill="#111827" textAnchor="middle">
        Frecuencia (Hz){log ? ' — escala logarítmica' : ''}
      </SvgText>
      <SvgText x={4} y={PAD.top + 4} style={{ fontSize: 7 }} fill="#111827">S</SvgText>
      {channels.map((ch) => (
        <G key={ch}>
          <Path d={path('g', ch)} stroke={channelColor(ch)} strokeWidth={1} fill="none" />
          <Path d={path('b', ch)} stroke={channelColor(ch)} strokeWidth={1} strokeDasharray="3,2" fill="none" />
        </G>
      ))}
    </Svg>
  );
};

export default MeasurementChartPdf;

import { createElement } from 'react';
import type { TaskReportData } from '../../types/report.types';

export async function generateTaskReportPdf(data: TaskReportData, generatedBy: string): Promise<void> {
  const [{ pdf }, { default: TaskReportDocument }] = await Promise.all([
    import('@react-pdf/renderer'),
    import('./TaskReportDocument'),

  saveBlob(blob, `reporte_${slug(data.task.name)}_${date}.pdf`);
}

      onClose();
    } catch (err) {
      const apiData = (err as { response?: { data?: unknown } })?.response?.data;
      toast.error(apiData ? getErrorMessage(apiData) : 'No se pudo generar el PDF');
    } finally {
      setBusy(false);
    }
  };

  return (
    <Dialog open onOpenChange={(open) => !open && !busy && onClose()}>
      <DialogContent className="max-w-md">
        <DialogHeader>
          <DialogTitle>Reporte PDF — {taskName}</DialogTitle>
        </DialogHeader>
        <p className="text-xs text-muted-foreground">Deja las fechas vacías para incluir todas las mediciones.</p>
        <div className="grid grid-cols-2 gap-3">
          <label className="space-y-1 text-xs text-muted-foreground">
            Desde
            <input type="date" value={from} onChange={(e) => setFrom(e.target.value)} className={inputCls} />
          </label>
          <label className="space-y-1 text-xs text-muted-foreground">
            Hasta
            <input type="date" value={to} onChange={(e) => setTo(e.target.value)} className={inputCls} />
          </label>
        </div>
        <div className="flex justify-end gap-2 pt-2">
          <KronosButton variant="outline" size="sm" onClick={onClose} disabled={busy}>Cancelar</KronosButton>
          <KronosButton
            size="sm"
            onClick={generate}
            disabled={busy}
            leftIcon={busy ? <Loader2 className="size-3.5 animate-spin" /> : <FileText className="size-3.5" />}
          >
            {busy ? 'Generando…' : 'Generar PDF'}
          </KronosButton>
        </div>
      </DialogContent>
    </Dialog>
  );
};

export default ReportOptionsModal;
```
